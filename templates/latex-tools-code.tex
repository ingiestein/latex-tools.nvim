% latex-tools-code.tex
% Project-local Markdown-like code fences (default backend: tcolorbox + listings).
% Copied beside each course-aware document; edit freely for that project.
%
% API (backend-agnostic — same surface in latex-tools-code-minted.tex):
%   \begin{mdcode}{Python} ... \end{mdcode}   % Lang must be a listings language name
%   \begin{python} ... \end{python}            % thin wrappers (preferred)
%   \begin{rcode} ... \end{rcode}
%   \begin{sql} ... \end{sql}
%   \begin{markdown} ... \end{markdown}
%   \begin{plaintext} ... \end{plaintext}
%   \begin{javascript} ... \end{javascript}
%   \begin{typescript} ... \end{typescript}
%   \begin{java} ... \end{java}
%   \begin{ccode} ... \end{ccode}             % C (env name avoids bare \begin{c})
%   \begin{cpp} ... \end{cpp}
%   \begin{csharp} ... \end{csharp}
%   \begin{rust} ... \end{rust}
%   \begin{json} ... \end{json}
%   \begin{go} ... \end{go}
%   \begin{bash} ... \end{bash}
%
% Note: fence envs intentionally have no optional [...] args. tcolorbox + listings
% treats # in the body as a macro parameter when the env is defined with #1.
% For line numbers, edit listing options below or add a project-local variant.
% Langs without a solid listings lexer (typescript, rust, json, go) use base style only.
%
% Opt-in minted+Pygments: in the parent, swap to
%   \input{latex-tools-code-minted}
% (requires pygmentize on PATH and compile with -shell-escape).

\usepackage{xcolor}
\usepackage{listings}
\usepackage[most]{tcolorbox}

% --- listings language skins (known langs); unknown-lang wrappers use latextoolstext ---
\lstdefinestyle{latextoolsbase}{
  basicstyle=\ttfamily\small,
  showstringspaces=false,
  breaklines=true,
  breakatwhitespace=false,
  tabsize=2,
  columns=fullflexible,
  keepspaces=true,
  numbers=none
}

  style=latextoolsbase,
  language=Python,
  keywordstyle=\color{blue!70!black}\bfseries,
  stringstyle=\color{green!40!black},

}

% --- shared Markdown / GitHub-inspired chrome (keep in sync with minted companion) ---

    enhanced,
    breakable,
    colback=gray!6,

    boxrule=0.4pt,
    arc=2pt,
    left=6pt,
    right=6pt,

    colbacktitle=gray!18,
    toptitle=2pt,
    bottomtitle=2pt,
    lefttitle=6pt,
    righttitle=6pt,
  },
}

% Generic fence: \begin{mdcode}{Lang} ... \end{mdcode}
% Lang should be a listings language name (Python, R, SQL, Java, ...) when known.
% No optional args — keeps # (Python comments, Markdown headings) legal in the body.
\DeclareTCBListing{mdcode}{ m }{%
  latextoolscode,
  title=#1,
  listing only,
  listing options={style=latextoolsbase,language=#1},
}
